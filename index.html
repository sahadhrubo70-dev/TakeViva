let questions = []; let index = 0; let score = 0; async function startInterview() { const file = document.getElementById("fileInput").files[0]; if (!file) { alert("Upload a file first"); return; } const text = await file.text(); const res = await fetch("/api/generate", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ text }) }); const data = await res.json(); if (data.error) { alert(data.error); return; } // FIX: proper split questions = data.questions .split("\n") .filter(q => q.trim() !== ""); index = 0; score = 0; showQuestion(); } function showQuestion() { document.getElementById("question").innerText = questions[index]; speak(questions[index]); } function nextQ() { let answer = document.getElementById("answer").value; if (answer.trim().length > 10) { score++; } document.getElementById("answer").value = ""; index++; if (index < questions.length) { showQuestion(); } else { document.getElementById("result").innerText = `Interview Done!\nScore: ${score}/${questions.length}`; } } function speak(text) { const msg = new SpeechSynthesisUtterance(text); window.speechSynthesis.speak(msg); }
